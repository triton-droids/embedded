cmake_minimum_required(VERSION 3.16)
project(experimental_publish_probe LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rcl REQUIRED)
find_package(rmw REQUIRED)
find_package(motor_control_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(zero_copy_capability zero_copy_capability.cpp)
target_compile_features(zero_copy_capability PRIVATE cxx_std_17)
ament_target_dependencies(zero_copy_capability rclcpp sensor_msgs std_msgs motor_control_interfaces)
add_executable(imu_bench imu_bench.cpp)
target_compile_features(imu_bench PRIVATE cxx_std_17)
ament_target_dependencies(imu_bench rclcpp sensor_msgs)
add_library(publish_boundary SHARED publish_boundary.cpp)
target_compile_features(publish_boundary PRIVATE cxx_std_17)
ament_target_dependencies(publish_boundary rcl rmw)
target_link_libraries(publish_boundary dl)
add_library(publish_wait SHARED publish_boundary.cpp)
target_compile_features(publish_wait PRIVATE cxx_std_17)
target_compile_definitions(publish_wait PRIVATE ENABLE_WAIT_STATS=1)
ament_target_dependencies(publish_wait rcl rmw)
target_link_libraries(publish_wait dl)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(fastrtps REQUIRED CONFIG)
add_executable(writer_qos writer_qos.cpp)
target_compile_features(writer_qos PRIVATE cxx_std_17)
ament_target_dependencies(writer_qos rclcpp sensor_msgs rmw_fastrtps_shared_cpp)
target_link_libraries(writer_qos fastrtps)
add_library(publish_failure SHARED publish_failure.cpp)
target_compile_features(publish_failure PRIVATE cxx_std_17)
ament_target_dependencies(publish_failure rcl rmw)
target_link_libraries(publish_failure dl)
add_library(publish_locks SHARED publish_boundary.cpp)
target_compile_features(publish_locks PRIVATE cxx_std_17)
target_compile_definitions(publish_locks PRIVATE ENABLE_MUTEX_TRACE=1)
ament_target_dependencies(publish_locks rcl rmw)
target_link_libraries(publish_locks dl)
